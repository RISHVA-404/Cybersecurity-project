========================================================================
CISCO PACKET TRACER CONFIGURATION COMMANDS
========================================================================
Instructions: 
1. Build the topology in Packet Tracer as defined in your manual.
2. Copy and paste these CLI commands into the respective devices.

------------------------------------------------------------------------
1. CORE-SW1 (Core Switch - VLAN Creation & Trunking)
------------------------------------------------------------------------
enable
configure terminal

! Create VLANs
vlan 10
 name FACULTY
vlan 20
 name APP_A
vlan 30
 name APP_B
vlan 40
 name DATABASE
vlan 50
 name DMZ
vlan 99
 name MANAGEMENT
exit

! Configure Trunk to DC-R1 (Assuming connected to Gig0/1)
interface gigabitEthernet 0/1
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,50,99
exit
! Configure Trunk to ACCESS-SW2 (Assuming connected to Gig0/2)
interface gigabitEthernet 0/2
 switchport mode trunk
exit
end
write memory

------------------------------------------------------------------------
2. ACCESS-SW2 (Access Switch - Access Ports)
------------------------------------------------------------------------
enable
configure terminal

! Create VLANs locally on this switch
vlan 10
 name FACULTY
vlan 20
 name APP_A
vlan 30
 name APP_B
vlan 40
 name DATABASE
vlan 50
 name DMZ
vlan 99
 name MANAGEMENT
exit

! Configure Trunk to CORE-SW1 (Assuming connected to Gig0/1)
interface gigabitEthernet 0/1
 switchport mode trunk
exit

! Configure Access Ports (Adjust port numbers based on your exact wiring)
interface fastEthernet 0/1
 switchport mode access
 switchport access vlan 10
! (Faculty PC 1)

interface fastEthernet 0/2
 switchport mode access
 switchport access vlan 10
! (Faculty PC 2)

interface fastEthernet 0/3
 switchport mode access
 switchport access vlan 20
! (App A Server)

interface fastEthernet 0/4
 switchport mode access
 switchport access vlan 30
! (App B Server)

interface fastEthernet 0/5
 switchport mode access
 switchport access vlan 40
! (Database Server)

interface fastEthernet 0/6
 switchport mode access
 switchport access vlan 50
! (DMZ Server)

interface fastEthernet 0/7
 switchport mode access
 switchport access vlan 99
! (Management PC)

end
write memory

------------------------------------------------------------------------
3. DC-R1 (Data Center Router - Inter-VLAN Routing, ACLs, SSH)
------------------------------------------------------------------------
enable
configure terminal

! Enable Physical Interface connected to CORE-SW1 (Assuming Gig0/0)
interface gigabitEthernet 0/0
 no shutdown
exit

! Configure Sub-interfaces for Inter-VLAN Routing (Router-on-a-Stick)
interface gigabitEthernet 0/0.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
exit

interface gigabitEthernet 0/0.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
exit

interface gigabitEthernet 0/0.30
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
exit

interface gigabitEthernet 0/0.40
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0
exit

interface gigabitEthernet 0/0.50
 encapsulation dot1Q 50
 ip address 192.168.50.1 255.255.255.0
exit

interface gigabitEthernet 0/0.99
 encapsulation dot1Q 99
 ip address 192.168.99.1 255.255.255.0
exit

! Configure Connection to Cloud Router (Assuming Gig0/1)
interface gigabitEthernet 0/1
 ip address 200.1.1.1 255.255.255.252
 no shutdown
exit

! Static Route to Cloud Network
ip route 172.16.10.0 255.255.255.0 200.1.1.2

! --- CONFIGURE ACLs ---

! ACL 100: Application Isolation (Block App A to App B)
access-list 100 deny ip 192.168.20.0 0.0.0.255 192.168.30.0 0.0.0.255
access-list 100 permit ip any any
! Apply to App A Interface
interface gigabitEthernet 0/0.20
 ip access-group 100 in
exit

! ACL 110: Protect Database (Block Faculty to Database)
access-list 110 deny ip 192.168.10.0 0.0.0.255 192.168.40.0 0.0.0.255
access-list 110 permit ip any any
! Apply to Faculty Interface
interface gigabitEthernet 0/0.10
 ip access-group 110 in
exit

! ACL 120: Block Cloud to Database
access-list 120 deny ip 172.16.10.0 0.0.0.255 192.168.40.0 0.0.0.255
access-list 120 permit ip any any
! Apply to Cloud WAN Interface
interface gigabitEthernet 0/1
 ip access-group 120 in
exit

! --- CONFIGURE SECURE MANAGEMENT (SSH) ---
ip domain-name smartcollege.local
username admin secret Cisco123
crypto key generate rsa
! (When prompted, enter 1024 for modulus)

! ACL to restrict VTY/SSH access to Management VLAN only
access-list 99 permit 192.168.99.0 0.0.0.255

line vty 0 4
 login local
 transport input ssh
 access-class 99 in
exit

end
write memory

------------------------------------------------------------------------
4. CLOUD-R1 (Cloud Router)
------------------------------------------------------------------------
enable
configure terminal

! Configure Connection to Data Center Router (Assuming Gig0/0)
interface gigabitEthernet 0/0
 ip address 200.1.1.2 255.255.255.252
 no shutdown
exit

! Configure Connection to Cloud App Network (Assuming Gig0/1)
interface gigabitEthernet 0/1
 ip address 172.16.10.1 255.255.255.0
 no shutdown
exit

! Static Route to Data Center Internal Networks
ip route 192.168.0.0 255.255.0.0 200.1.1.1

end
write memory
