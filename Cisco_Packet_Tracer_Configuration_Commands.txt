========================================================================
CISCO PACKET TRACER CONFIGURATION COMMANDS - NEXGEN CAMPUS
========================================================================
Instructions: 
1. Build the topology in Packet Tracer using the NexGen Campus names.
2. Copy and paste these CLI commands into the respective devices.

------------------------------------------------------------------------
1. NGC-CORE-SW (Core Switch - VLAN Creation & Trunking)
------------------------------------------------------------------------
enable
configure terminal

! Create VLANs
vlan 100
 name STAFF
vlan 110
 name APP_1
vlan 120
 name APP_2
vlan 130
 name DATABASE
vlan 140
 name DMZ
vlan 199
 name MANAGEMENT
exit

! Configure Trunk to NGC-CORE-RTR (Assuming Gig0/1)
interface gigabitEthernet 0/1
 switchport mode trunk
 switchport trunk allowed vlan 100,110,120,130,140,199
exit

! Configure Trunk to NGC-ACCESS-SW (Assuming Gig0/2)
interface gigabitEthernet 0/2
 switchport mode trunk
exit
end
write memory

------------------------------------------------------------------------
2. NGC-ACCESS-SW (Access Switch - Access Ports)
------------------------------------------------------------------------
enable
configure terminal

! Create VLANs locally on this switch
vlan 100
 name STAFF
vlan 110
 name APP_1
vlan 120
 name APP_2
vlan 130
 name DATABASE
vlan 140
 name DMZ
vlan 199
 name MANAGEMENT
exit

! Configure Trunk to NGC-CORE-SW (Assuming Gig0/1)
interface gigabitEthernet 0/1
 switchport mode trunk
exit

! Configure Access Ports
interface fastEthernet 0/1
 switchport mode access
 switchport access vlan 100
! (STAFF-PC1)

interface fastEthernet 0/2
 switchport mode access
 switchport access vlan 100
! (STAFF-PC2)

interface fastEthernet 0/3
 switchport mode access
 switchport access vlan 110
! (APP-SRV-1)

interface fastEthernet 0/4
 switchport mode access
 switchport access vlan 120
! (APP-SRV-2)

interface fastEthernet 0/5
 switchport mode access
 switchport access vlan 130
! (DATA-SRV)

interface fastEthernet 0/6
 switchport mode access
 switchport access vlan 140
! (PUBLIC-SRV)

interface fastEthernet 0/7
 switchport mode access
 switchport access vlan 199
! (ADMIN-PC)

end
write memory

------------------------------------------------------------------------
3. NGC-CORE-RTR (Main Data Center Router)
------------------------------------------------------------------------
enable
configure terminal

! Enable Physical Interface connected to NGC-CORE-SW (Assuming Gig0/0)
interface gigabitEthernet 0/0
 no shutdown
exit

! Configure Sub-interfaces (Router-on-a-Stick)
interface gigabitEthernet 0/0.100
 encapsulation dot1Q 100
 ip address 172.20.10.1 255.255.255.0
exit

interface gigabitEthernet 0/0.110
 encapsulation dot1Q 110
 ip address 172.20.11.1 255.255.255.0
exit

interface gigabitEthernet 0/0.120
 encapsulation dot1Q 120
 ip address 172.20.12.1 255.255.255.0
exit

interface gigabitEthernet 0/0.130
 encapsulation dot1Q 130
 ip address 172.20.13.1 255.255.255.0
exit

interface gigabitEthernet 0/0.140
 encapsulation dot1Q 140
 ip address 172.20.14.1 255.255.255.0
exit

interface gigabitEthernet 0/0.199
 encapsulation dot1Q 199
 ip address 172.20.99.1 255.255.255.0
exit

! Configure Connection to Cloud Router (Assuming Gig0/1)
interface gigabitEthernet 0/1
 ip address 10.0.0.1 255.255.255.252
 no shutdown
exit

! Static Route to Cloud Network
ip route 10.1.1.0 255.255.255.0 10.0.0.2

! --- CONFIGURE ACLs ---

! ACL 101: Block App 1 to App 2 (Application Isolation)
access-list 101 deny ip 172.20.11.0 0.0.0.255 172.20.12.0 0.0.0.255
access-list 101 permit ip any any
interface gigabitEthernet 0/0.110
 ip access-group 101 in
exit

! ACL 102: Block Staff to Database (Database Protection)
access-list 102 deny ip 172.20.10.0 0.0.0.255 172.20.13.0 0.0.0.255
access-list 102 permit ip any any
interface gigabitEthernet 0/0.100
 ip access-group 102 in
exit

! ACL 103: Block Cloud to Database (Cloud Security)
access-list 103 deny ip 10.1.1.0 0.0.0.255 172.20.13.0 0.0.0.255
access-list 103 permit ip any any
interface gigabitEthernet 0/1
 ip access-group 103 in
exit

! ACL 104: Block DMZ to Database (DMZ Isolation)
access-list 104 deny ip 172.20.14.0 0.0.0.255 172.20.13.0 0.0.0.255
access-list 104 permit ip any any
interface gigabitEthernet 0/0.140
 ip access-group 104 in
exit

! --- CONFIGURE SECURE MANAGEMENT (SSH) ---
ip domain-name nexgen.local
username admin secret cisco123
crypto key generate rsa
! (When prompted, enter 1024 for modulus)

! ACL 99: Restrict SSH access to Management VLAN
access-list 99 permit 172.20.99.0 0.0.0.255

line vty 0 4
 login local
 transport input ssh
 access-class 99 in
exit

end
write memory

------------------------------------------------------------------------
4. NGC-CLOUD-RTR (Simulated Cloud Gateway)
------------------------------------------------------------------------
enable
configure terminal

! Configure Connection to NGC-CORE-RTR (Assuming Gig0/0)
interface gigabitEthernet 0/0
 ip address 10.0.0.2 255.255.255.252
 no shutdown
exit

! Configure Connection to Cloud Server (Assuming Gig0/1)
interface gigabitEthernet 0/1
 ip address 10.1.1.1 255.255.255.0
 no shutdown
exit

! Static Route back to Data Center
ip route 172.20.0.0 255.255.0.0 10.0.0.1

end
write memory

------------------------------------------------------------------------
5. END DEVICE IP CONFIGURATION
------------------------------------------------------------------------
Configure these statically in the Packet Tracer GUI:

STAFF-PC1:      IP 172.20.10.10 / Mask 255.255.255.0 / Gateway 172.20.10.1
STAFF-PC2:      IP 172.20.10.11 / Mask 255.255.255.0 / Gateway 172.20.10.1
APP-SRV-1:      IP 172.20.11.10 / Mask 255.255.255.0 / Gateway 172.20.11.1
APP-SRV-2:      IP 172.20.12.10 / Mask 255.255.255.0 / Gateway 172.20.12.1
DATA-SRV:       IP 172.20.13.10 / Mask 255.255.255.0 / Gateway 172.20.13.1
PUBLIC-SRV:     IP 172.20.14.10 / Mask 255.255.255.0 / Gateway 172.20.14.1
ADMIN-PC:       IP 172.20.99.10 / Mask 255.255.255.0 / Gateway 172.20.99.1
CLOUD-SRV:      IP 10.1.1.10    / Mask 255.255.255.0 / Gateway 10.1.1.1
